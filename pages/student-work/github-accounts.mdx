export function AccountLinks() {
  const accountCSV = process.env.NEXT_PUBLIC_STUDENT_GITHUB_USERNAMES;

// If the environment variable is not set or empty, show nothing or a friendly message
if (!accountCSV || accountCSV.trim() === '') {
return null; // or return <p>No student GitHub accounts listed yet.</p> if you prefer a message
}

const accounts = accountCSV
.split(',')
.map((account) => account.trim())
.filter((account) => account.length > 0);

// If after cleaning there are still no accounts, don't render anything
if (accounts.length === 0) {
return null;
}

return (
<div>
{accounts.map((account, index) => (
<span key={account}>
{" "}
<a href={`https://github.com/${account}`} target="\_blank" rel="noopener noreferrer">
{account}
</a>
{index < accounts.length - 1 ? " |" : ""}
</span>
))}
</div>
);
}
